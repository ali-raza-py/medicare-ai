+        '/api/documents/upload',

+        headers=_auth_headers("owner@example.com"),
+    )

+    assert response.json() == {'document_id': document_id, 'deleted': True}
+    assert store.get(document_id) is None
+    assert client.get(
+        f'/api/documents/{document_id}',
